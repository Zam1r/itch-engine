cmake_minimum_required(VERSION 3.20)
project(itch_engine CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG")
option(ITCH_SANITIZE "Build with ASan+UBSan" OFF)
if(ITCH_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra)
include_directories(include src)

add_executable(itch_replay src/replay.cpp)
add_executable(itch_verify src/verify.cpp)

include(FetchContent)
FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.7.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(Catch2)
add_executable(tests tests/decoder_test.cpp tests/book_test.cpp tests/order_map_test.cpp)
target_link_libraries(tests PRIVATE Catch2::Catch2WithMain)
enable_testing()
add_test(NAME unit COMMAND tests)
